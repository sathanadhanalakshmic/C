#include <stdio.h>
#include <string.h>
int main()
{
   char str[100];
   int choice, pos, len, i, count;
   char ch;
   printf("Enter a string: ");
   scanf("%s",&str);
   len = strlen(str);
   printf("Enter your choice: ");
   scanf("%d", &choice);
   switch (choice)
   {
      case 1:
         printf("Enter a character to search: ");
         scanf(" %c", &ch);
         for (i = 0; i < len; i++) {
            if (str[i] == ch) {
               printf("Character found at position %d\n", i + 1);
               break;
            }
         }
         if (i == len) {
            printf("Character not found\n");
         }
         break;

      case 2:
         printf("Enter a position: ");
         scanf("%d", &pos);
         printf("Character at position %d is %c\n", pos, str[pos - 1]);
         break;

      case 3:
         printf("Enter a character to count: ");
         scanf(" %c", &ch);
         count = 0;
         for (i = 0; i < len; i++) {
            if (str[i] == ch) {
               count++;
            }
         }
         printf("Number of occurrences of %c: %d\n", ch, count);
         break;

      case 4:
         for (i = 0, count = 0; i < len / 2; i++) {
            if (str[i] != str[len - i - 1]) {
               count++;
            }
         }
         if (count == 0) {
            printf("String is a palindrome\n");
         } else {
            printf("String is not a palindrome\n");
         }
         break;
   }
}
