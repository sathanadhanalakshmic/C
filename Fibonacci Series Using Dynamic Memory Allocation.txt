include <stdio.h>
#include<stdlib.h>
int main()
{
   int n, i, *fib;
   printf("Enter the number of terms: ");
   scanf("%d", &n);
   fib = (int *)calloc(n,sizeof(int));
   *(fib+1)=1;
   for (i = 2; i<=n; i++)
   {
     *(fib+i)=*(fib+i-1)+*(fib+i-2);
   }
   printf("Fibonacci series: ");
   for (i = 0; i <=n-1; i++)
   {
      printf("%d ", fib[i]);
   }
}
