#include <stdio.h>
#include <string.h>
int main()
{
   char a[100],b[100];
   int i, j=0, count = 0;
   printf("Enter the paragraph: ");
   scanf("%s",&a);
   printf("Enter the substring: ");
   scanf("%s",&b);
  for(i=0;a[i]!='\0';i++)
  {
     if(a[i]==b[i])
        j++;
     else if(a[i]==b[0])
        j=1;
     else
        j=0;
     if(b[j]=='\0')
     {
        count++;
        j=0;
     }
  }

   printf("Number of occurrences of the substring: %d\n", count);
   return 0;
}
