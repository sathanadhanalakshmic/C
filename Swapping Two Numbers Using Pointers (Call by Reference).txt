#include<stdio.h>
void swap(int *p1,int *p2);
int main()
{
   int x,y;
   printf("Enter the values for x and y:");
   scanf("%d,%d",&x,&y);
   printf("Before swapping x=%d,y=%d\n",x,y);
   swap(&x,&y);
   printf("After swapping x=%d,y=%d\n",x,y);
}
void swap(int *p1,int *p2)
{
   int temp;
   temp=*p1;
   *p1=*p2;
   *p2=temp;
}
