#include <stdio.h>
#include <stdlib.h>
int main() {
   int n, i;
   printf("Enter the number of elements: ");
   scanf("%d", &n);
   int *arr = (int *)malloc(n * sizeof(int));
   printf("Enter the elements:\n");
   for (i = 0; i < n; i++) {
      scanf("%d", &arr[i]);
   }
   printf("Elements in reverse order: ");
   for (i = n - 1; i >= 0; i--) {
      printf("%d ", arr[i]);
   }
   free(arr);
   return 0;

}
