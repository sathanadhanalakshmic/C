#include<stdio.h>
int main()
{
   char std[50],ad1[50],ad2[50];
   int n,i,*p;
   printf("Enter the number of students:");
   scanf("%d",&n);
   for(i=0;i<n;i++)
   {
      std[i]=i+1;
      printf("\nThe roll number of %d student:%d",i+1,std[i]);
   }
   for(i=0;i<n;i++)
   {
      *p=std[i];
      if((i+1)%2==0)
      {
         ad1[i]=*p;
         printf("\n The first staff advisor's student with roll number:%d",ad1[i]);
         p++;
      }
      else
      {
         ad2[i]=*p;
         printf("\n The second staff advisor's student with roll number:%d",ad2[i]);
      }
   }
   return 0;
}
